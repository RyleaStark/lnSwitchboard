Reverse Proxy

Serve LNURL + Nostr

Copy the /.well-known/lnurlp/ and /.well-known/nostr.json paths to each domain and point them at this service. 
server {
    listen 443 ssl;
    server_name your.domain;

    # LNURL-pay proxy
    location ~ ^/\.well-known/lnurlp/[a-z0-9._+-]+$ {
        proxy_pass http://127.0.0.1:22121$request_uri;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $remote_addr;
    }

    # NIP-05 nostr identity lookup
    location = /.well-known/nostr.json {
        proxy_pass http://127.0.0.1:22121$request_uri;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $remote_addr;
    }
}
